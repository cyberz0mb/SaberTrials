cmake_minimum_required(VERSION 3.21)
project(SaberTrialsAntiScriptTests LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(src "${CMAKE_CURRENT_SOURCE_DIR}/../../development-source/anti-script")
add_subdirectory(../../development-source/staff staff-core)
add_library(as_core "${src}/detection.cpp" "${src}/ledger.cpp")
target_link_libraries(as_core PUBLIC st_sqlite)
target_include_directories(as_core PUBLIC "${src}")
add_executable(as_tests checks.cpp)
target_link_libraries(as_tests PRIVATE as_core)
foreach(suite cfg behavior ledger)
  add_test(NAME as_${suite} COMMAND as_tests ${suite})
endforeach()
add_executable(as_adapter adapter.cpp "${src}/server_hooks.cpp")
target_include_directories(as_adapter PRIVATE stubs ../../development-source/accounts/codemp/accounts)
target_link_libraries(as_adapter PRIVATE as_core st_staff_core)
add_test(NAME as_server_lifecycle COMMAND as_adapter)

if(DEFINED OPENJK_SOURCE)
  add_executable(as_parser parser.cpp "${OPENJK_SOURCE}/codemp/qcommon/cmd.cpp" "${src}/client_trace.cpp")
  target_include_directories(as_parser BEFORE PRIVATE parser)
  target_link_libraries(as_parser PRIVATE as_core)
  add_test(NAME as_engine_command_buffer COMMAND as_parser)
endif()
