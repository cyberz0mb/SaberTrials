cmake_minimum_required(VERSION 3.16)
project(TranslatorTests CXX)
enable_testing()
add_executable(translator_tests runtime.cpp)
target_compile_features(translator_tests PRIVATE cxx_std_11)
target_include_directories(translator_tests PRIVATE ../../development-source/translator)
if(MSVC)
  target_compile_options(translator_tests PRIVATE /UNDEBUG /utf-8)
endif()
add_test(NAME translator_contracts COMMAND translator_tests "${CMAKE_CURRENT_LIST_DIR}/../../dist/translator-02/runtime")
add_executable(chat_text_tests text.cpp)
target_compile_features(chat_text_tests PRIVATE cxx_std_11)
target_include_directories(chat_text_tests PRIVATE ../../development-source/translator ../../dist/translator-02/engine/codemp/qcommon ../../dist/translator-02/engine/codemp)
if(MSVC)
  target_compile_options(chat_text_tests PRIVATE /UNDEBUG /utf-8)
endif()
add_test(NAME unicode_chat_contracts COMMAND chat_text_tests "${CMAKE_CURRENT_LIST_DIR}/../../dist/translator-02/runtime/fonts")
