cmake_minimum_required(VERSION 3.16)
project(SaberTrialsHybridTests C CXX)
enable_testing()
set(HYBRID "${CMAKE_CURRENT_SOURCE_DIR}/../../development-source/hybrid-styles")
add_executable(hybrid_rules rules.c "${HYBRID}/st_hybrid.c")
target_include_directories(hybrid_rules PRIVATE "${HYBRID}")
add_test(NAME hybrid_rules COMMAND hybrid_rules)
add_executable(hybrid_server_loadout server_loadout.c "${HYBRID}/st_hybrid.c")
target_include_directories(hybrid_server_loadout PRIVATE "${HYBRID}" "${CMAKE_CURRENT_SOURCE_DIR}/../../development-source/combat-foundation")
add_test(NAME hybrid_server_loadout COMMAND hybrid_server_loadout)
add_executable(hybrid_loadout loadout.c "${HYBRID}/st_hybrid.c")
target_include_directories(hybrid_loadout PRIVATE "${HYBRID}")
add_test(NAME hybrid_loadout COMMAND hybrid_loadout)
set(COMBAT "${CMAKE_CURRENT_SOURCE_DIR}/../../development-source/combat-foundation")
add_executable(hybrid_adapter adapter.c "${HYBRID}/st_hybrid.c" "${COMBAT}/st_combat_core.c")
target_include_directories(hybrid_adapter PRIVATE "${HYBRID}" "${COMBAT}")
add_test(NAME hybrid_adapter COMMAND hybrid_adapter)
if(ST_HYBRID_ENGINE)
 get_filename_component(CANDIDATE "${ST_HYBRID_ENGINE}" DIRECTORY)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_damage_fixture.py"
  "${CANDIDATE}" "${CMAKE_CURRENT_BINARY_DIR}/damage_fixture.cpp" RESULT_VARIABLE DAMAGE_RESULT)
 if(NOT DAMAGE_RESULT EQUAL 0)
  message(FATAL_ERROR "Production hybrid damage extraction failed")
 endif()
 execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_reaction_fixture.py"
  "${CANDIDATE}" "${CMAKE_CURRENT_BINARY_DIR}/reaction_fixture.inc" RESULT_VARIABLE REACTION_RESULT)
 if(NOT REACTION_RESULT EQUAL 0)
  message(FATAL_ERROR "Production saber reaction extraction failed")
 endif()
 add_executable(hybrid_reactions reactions.c "${HYBRID}/st_hybrid.c" "${COMBAT}/st_combat_core.c")
 target_include_directories(hybrid_reactions PRIVATE "${HYBRID}" "${COMBAT}" "${CMAKE_CURRENT_BINARY_DIR}")
 add_test(NAME hybrid_production_reactions COMMAND hybrid_reactions)
 target_include_directories(hybrid_adapter PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 target_compile_definitions(hybrid_adapter PRIVATE STH_CHAIN_FIXTURE)
 add_executable(hybrid_damage "${CMAKE_CURRENT_BINARY_DIR}/damage_fixture.cpp" "${HYBRID}/st_hybrid.c")
 target_include_directories(hybrid_damage PRIVATE "${HYBRID}" "${CANDIDATE}/game/codemp")
 add_test(NAME hybrid_production_damage COMMAND hybrid_damage)
 configure_file("${ST_HYBRID_ENGINE}/shared/qcommon/q_version.h.in" qcommon/q_version.h)
 foreach(which hybrid baseline)
  if(which STREQUAL "baseline")
   if(NOT ST_BASELINE_ENGINE)
    continue()
   endif()
   set(ENGINE "${ST_BASELINE_ENGINE}")
  else()
   set(ENGINE "${ST_HYBRID_ENGINE}")
  endif()
  add_executable(${which}_protocol protocol.cpp "${ENGINE}/codemp/qcommon/msg.cpp"
   "${ENGINE}/codemp/qcommon/huffman.cpp" "${ENGINE}/codemp/qcommon/q_shared.c"
   "${ENGINE}/shared/qcommon/q_math.c" "${ENGINE}/shared/qcommon/q_string.c")
  target_include_directories(${which}_protocol PRIVATE "${ENGINE}/codemp" "${ENGINE}/shared" "${CMAKE_CURRENT_BINARY_DIR}")
  set_source_files_properties("${ENGINE}/codemp/qcommon/q_shared.c" "${ENGINE}/shared/qcommon/q_math.c"
   "${ENGINE}/shared/qcommon/q_string.c" PROPERTIES LANGUAGE CXX)
  if(which STREQUAL "baseline")
   target_compile_definitions(${which}_protocol PRIVATE STH_PROTOCOL_BASELINE)
  endif()
  if(MSVC)
   target_compile_options(${which}_protocol PRIVATE /Gy)
   target_link_options(${which}_protocol PRIVATE /OPT:REF)
  endif()
  add_test(NAME ${which}_snapshot COMMAND ${which}_protocol)
 endforeach()
endif()
