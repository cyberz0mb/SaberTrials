cmake_minimum_required(VERSION 3.20)
project(st_controls_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
add_executable(controls_core core.cpp)
target_include_directories(controls_core PRIVATE ../../development-source/controls)
enable_testing()
add_test(NAME controls_core COMMAND controls_core)
add_executable(controls_controller controller.cpp)
target_include_directories(controls_controller PRIVATE ../../development-source/controls)
add_test(NAME controls_controller COMMAND controls_controller)
if(ST_CONTROLLER_ENGINE)
    # Exercise the combined release's movement code in the same simulated SDL fixture.
    add_executable(controller_release controller.cpp)
    target_compile_definitions(controller_release PRIVATE ST_RELEASE_CONTROLLER)
    target_include_directories(controller_release PRIVATE "${ST_CONTROLLER_ENGINE}/codemp" ../../development-source/controls)
    add_test(NAME controller_release COMMAND controller_release)
endif()
