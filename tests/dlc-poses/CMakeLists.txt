cmake_minimum_required(VERSION 3.20)
project(DlcPoseTests CXX)
enable_testing()
set(ST_ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/../../dist/dlc-pose-01/engine" CACHE PATH "Engine source")
set(ST_POSE_HEADERS "${ST_ENGINE}/codemp/ghoul2" CACHE PATH "Compatibility headers")
set(ST_FIXTURES "${CMAKE_CURRENT_SOURCE_DIR}/../../dist/dlc-pose-01/fixtures" CACHE PATH "Read-only model fixtures")
add_executable(dlc_pose_tests models.cpp)
target_compile_features(dlc_pose_tests PRIVATE cxx_std_11)
file(STRINGS "${ST_ENGINE}/codemp/rd-rend2/tr_local.h" ST_RENDERER_BUDGET REGEX "^#define MAX_G2_BONES ")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/renderer_budget.h" "${ST_RENDERER_BUDGET}\n")
target_include_directories(dlc_pose_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(dlc_pose_tests PRIVATE "${ST_POSE_HEADERS}" "${ST_ENGINE}/codemp/ghoul2" "${ST_ENGINE}/codemp" "${ST_ENGINE}/shared")
foreach(model martian alien spacewolves akaza)
  add_test(NAME extend_${model} COMMAND dlc_pose_tests extend "${ST_FIXTURES}/${model}.gla" "${ST_FIXTURES}/canonical.gla")
endforeach()
add_test(NAME incompatible_limb_rig_keeps_own_frames COMMAND dlc_pose_tests incompatible "${ST_FIXTURES}/hk47.gla" "${ST_FIXTURES}/canonical.gla")
add_test(NAME alien_mesh_accepted_by_rend2 COMMAND dlc_pose_tests mesh "${ST_FIXTURES}/alien_mesh.glm")
add_executable(dlc_body_math ../handedness/gla.cpp)
target_compile_features(dlc_body_math PRIVATE cxx_std_11)
target_include_directories(dlc_body_math PRIVATE "${ST_POSE_HEADERS}" "${ST_ENGINE}/codemp/ghoul2" "${ST_ENGINE}/codemp" "${ST_ENGINE}/shared")
add_test(NAME canonical_hand_transforms COMMAND dlc_body_math "${ST_FIXTURES}/canonical.gla")
add_test(NAME martian_hand_transforms COMMAND dlc_body_math "${ST_FIXTURES}/martian.gla")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py"
  "${ST_ENGINE}/codemp/rd-vanilla/G2_API.cpp" "${CMAKE_CURRENT_BINARY_DIR}/setter.inc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(dlc_frame_range range.cpp)
target_compile_features(dlc_frame_range PRIVATE cxx_std_11)
target_include_directories(dlc_frame_range PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ST_POSE_HEADERS}" "${ST_ENGINE}/codemp/ghoul2" "${ST_ENGINE}/codemp" "${ST_ENGINE}/shared")
add_test(NAME missing_frames_preserve_pose COMMAND dlc_frame_range)
foreach(model martian alien akaza canonical)
  add_test(NAME mirror_${model} COMMAND dlc_pose_tests mirror "${ST_FIXTURES}/${model}.gla")
endforeach()
