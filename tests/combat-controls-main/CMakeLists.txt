cmake_minimum_required(VERSION 3.16)
project(SaberTrialsMainCombatChecks C CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT ST_CANDIDATE)
 message(FATAL_ERROR "ST_CANDIDATE is required")
endif()
set(ST_COMBAT_BASELINE "${CMAKE_CURRENT_SOURCE_DIR}/../../dist/multiteam-01/game" CACHE PATH "Input game source before combat controls")
set(GAME "${ST_CANDIDATE}/game/codemp/game")
set(ENGINE "${ST_CANDIDATE}/engine")
set(FIXTURES "${CMAKE_CURRENT_BINARY_DIR}/production")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py" "${ST_CANDIDATE}" "${FIXTURES}" "${ST_COMBAT_BASELINE}" RESULT_VARIABLE prepared)
if(NOT prepared EQUAL 0)
 message(FATAL_ERROR "Production fixture extraction failed")
endif()
add_executable(main_feint "${FIXTURES}/feint_adapter.c" "${GAME}/st_combat_core.c" "${GAME}/st_combat_wire.c")
add_executable(main_block "${FIXTURES}/manual_block.c" "${GAME}/st_combat_core.c")
add_executable(main_guard "${FIXTURES}/held_guard.c")
add_executable(main_combat_sync sync.c)
target_compile_definitions(main_guard PRIVATE ST_GUARD_POSE_PRESENT)
foreach(target main_feint main_block main_guard main_combat_sync)
 target_include_directories(${target} PRIVATE "${GAME}" "${FIXTURES}")
 add_test(NAME ${target} COMMAND ${target})
endforeach()
configure_file("${ENGINE}/shared/qcommon/q_version.h.in" qcommon/q_version.h)
add_executable(main_guard_wire "${FIXTURES}/manual_block_wire.cpp"
 "${ENGINE}/codemp/qcommon/msg.cpp" "${ENGINE}/codemp/qcommon/huffman.cpp"
 "${ENGINE}/codemp/qcommon/q_shared.c" "${ENGINE}/shared/qcommon/q_math.c" "${ENGINE}/shared/qcommon/q_string.c")
target_include_directories(main_guard_wire PRIVATE "${ENGINE}/codemp" "${ENGINE}/shared" "${CMAKE_CURRENT_BINARY_DIR}" "${GAME}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(main_guard_wire BEFORE PRIVATE "${GAME}/..")
set_source_files_properties("${ENGINE}/codemp/qcommon/q_shared.c" "${ENGINE}/shared/qcommon/q_math.c" "${ENGINE}/shared/qcommon/q_string.c" PROPERTIES LANGUAGE CXX)
if(MSVC)
 target_compile_options(main_guard_wire PRIVATE /Gy)
 target_link_options(main_guard_wire PRIVATE /OPT:REF)
endif()
add_test(NAME main_guard_wire COMMAND main_guard_wire)
