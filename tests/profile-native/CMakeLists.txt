cmake_minimum_required(VERSION 3.20)
project(st_profile_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
add_executable(profile_core core.cpp)
target_include_directories(profile_core PRIVATE ../../development-source/profile)
if(MSVC)
  target_compile_options(profile_core PRIVATE /UNDEBUG)
else()
  target_compile_options(profile_core PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME profile_core COMMAND profile_core)
