cmake_minimum_required(VERSION 3.16)
project(st_combat_foundation C)
enable_testing()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../development-source/combat-foundation")
add_library(st_combat_core STATIC "${CORE}/st_combat_core.c")
target_include_directories(st_combat_core PUBLIC "${CORE}")
if(MSVC)
 target_compile_options(st_combat_core PRIVATE /W4 /WX)
else()
 target_compile_options(st_combat_core PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_executable(feint_checks feint_checks.c)
target_link_libraries(feint_checks PRIVATE st_combat_core)
add_test(NAME feint_controller COMMAND feint_checks)
add_executable(style_checks style_checks.c)
target_link_libraries(style_checks PRIVATE st_combat_core)
add_test(NAME style_catalog COMMAND style_checks)
add_executable(wire_checks wire_checks.c "${CORE}/st_combat_wire.c")
target_link_libraries(wire_checks PRIVATE st_combat_core)
add_test(NAME combat_snapshot_codec COMMAND wire_checks)
add_executable(combat_catalog catalog_export.c)
target_link_libraries(combat_catalog PRIVATE st_combat_core)
if(ST_DAMAGE_FIXTURES)
 add_executable(damage_checks damage_checks.c)
 target_include_directories(damage_checks PRIVATE "${ST_DAMAGE_FIXTURES}")
 if(NOT MSVC)
  target_link_libraries(damage_checks PRIVATE m)
 endif()
 add_test(NAME accepted_damage_ramp COMMAND damage_checks)
endif()
if(ST_MOVEMENT_FIXTURES)
 add_executable(movement_checks movement_checks.c)
 target_include_directories(movement_checks PRIVATE "${ST_MOVEMENT_FIXTURES}")
 if(NOT MSVC)
  target_link_libraries(movement_checks PRIVATE m)
 endif()
 add_test(NAME staff_prediction_parity COMMAND movement_checks)
endif()
