cmake_minimum_required(VERSION 3.21)
get_filename_component(ST_PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ST_DEPS "${ST_PROJECT_ROOT}/dist/staff-deps" CACHE PATH "Verified staff dependency extraction")
add_library(st_sodium STATIC IMPORTED GLOBAL)
if(WIN32)
set_target_properties(st_sodium PROPERTIES IMPORTED_LOCATION "${ST_DEPS}/libsodium-stable/bin/x64/Release/v145/static/libsodium.lib")
target_include_directories(st_sodium INTERFACE "${ST_DEPS}/libsodium-stable/src/libsodium/include")
target_compile_definitions(st_sodium INTERFACE SODIUM_STATIC)
else()
 set(ST_SODIUM_PREFIX "${ST_DEPS}/sodium-linux" CACHE PATH "Pinned Linux libsodium prefix")
 set_target_properties(st_sodium PROPERTIES IMPORTED_LOCATION "${ST_SODIUM_PREFIX}/lib/libsodium.a")
 target_include_directories(st_sodium INTERFACE "${ST_SODIUM_PREFIX}/include")
endif()
add_library(st_sqlite STATIC "${ST_DEPS}/sqlite-amalgamation-3540000/sqlite3.c")
target_include_directories(st_sqlite PUBLIC "${ST_DEPS}/sqlite-amalgamation-3540000")
target_compile_definitions(st_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
add_library(st_staff_core STATIC codemp/staff/st_staff_identity.cpp codemp/staff/st_staff_store.cpp codemp/staff/st_staff_protocol.cpp codemp/staff/st_staff_wire.cpp)
target_include_directories(st_staff_core PUBLIC codemp/staff)
target_compile_features(st_staff_core PUBLIC cxx_std_17)
target_compile_definitions(st_staff_core PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(st_staff_core PUBLIC st_sodium st_sqlite)
if(WIN32)
 target_link_libraries(st_staff_core PUBLIC crypt32 bcrypt advapi32 ws2_32 user32)
else()
 target_sources(st_staff_core PRIVATE codemp/staff/st_private_linux.cpp)
 find_package(Threads REQUIRED)
 target_link_libraries(st_staff_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
endif()
